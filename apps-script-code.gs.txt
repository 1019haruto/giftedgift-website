var SHEET_NAME = 'Submissions';
var PLAN_SHEET_NAME = 'Plans';
var SECRET = 'giftsecret26'; // ← 必ず自分だけの文字列に変更してください（config.jsのADMIN_SECRETと同じにする）

function doPost(e) {
  var data = JSON.parse(e.postData.contents);

  if (data.type === 'plan') {
    var planSheet = getOrCreateSheet(PLAN_SHEET_NAME);
    planSheet.appendRow([
      new Date(),
      data.email || '',
      data.title || '',
      data.page || '',
      JSON.stringify(data.schedule || []),
      data.status || 'ご提案中'
    ]);
    return jsonOutput({ result: 'ok' });
  }

  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME);
  sheet.appendRow([
    new Date(),
    data.page || '',
    data.email || '',
    JSON.stringify(data.categories || [], null, 0),
    data.notes || ''
  ]);

  if (data.email && data.code) {
    sendConfirmationEmail(data.email, data.code);
  }

  return jsonOutput({ result: 'ok' });
}

function sendConfirmationEmail(email, code) {
  var subject = '【GIFTED×GIFT】ご相談内容を受け付けました';
  var body =
    email + ' 様\n\n' +
    'この度はGIFTED×GIFTへご相談いただき、誠にありがとうございます。\n' +
    '内容を確認のうえ、担当より改めてご連絡させていただきます。\n\n' +
    '▼確認コード（次回のマイページログインにご利用いただけます）\n' +
    code + '\n\n' +
    '※このメールは送信専用です。ご返信いただいてもお答えできませんのでご了承ください。';

  GmailApp.sendEmail(email, subject, body);
}

function doGet(e) {
  if (e.parameter.action === 'myplans') {
    var email = (e.parameter.email || '').toLowerCase();
    if (!email) return jsonOutput({ error: 'email required' });

    var planSheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(PLAN_SHEET_NAME);
    var items = [];

    if (planSheet) {
      var planRows = planSheet.getDataRange().getValues();
      for (var p = 0; p < planRows.length; p++) {
        if (String(planRows[p][1]).toLowerCase() === email) {
          items.push({
            date: planRows[p][0],
            email: planRows[p][1],
            title: planRows[p][2],
            page: planRows[p][3],
            schedule: planRows[p][4],
            status: planRows[p][5]
          });
        }
      }
      items.reverse();
    }

    return jsonOutput(items);
  }

  if (!e.parameter.key || e.parameter.key !== SECRET) {
    return jsonOutput({ error: 'unauthorized' });
  }

  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME);
  var rows = sheet.getDataRange().getValues();
  var items = [];

  for (var i = 0; i < rows.length; i++) {
    items.push({
      date: rows[i][0],
      page: rows[i][1],
      email: rows[i][2],
      categories: rows[i][3],
      notes: rows[i][4]
    });
  }

  items.reverse();

  return jsonOutput(items);
}

function getOrCreateSheet(name) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName(name);
  if (!sheet) sheet = ss.insertSheet(name);
  return sheet;
}

function jsonOutput(obj) {
  return ContentService
    .createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}
