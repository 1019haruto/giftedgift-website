var SHEET_NAME = 'Submissions';
var PLAN_SHEET_NAME = 'Plans';
var SECRET = 'giftsecret26'; // ← 必ず自分だけの文字列に変更してください（config.jsのADMIN_SECRETと同じにする）

function doPost(e) {
  var data = JSON.parse(e.postData.contents);

  if (data.type === 'plan') {
    var planSheet = getOrCreateSheet(PLAN_SHEET_NAME);
    planSheet.appendRow([
      new Date(),
      data.email || '',
      data.title || '',
      data.page || '',
      JSON.stringify(data.schedule || []),
      data.status || 'ご提案中'
    ]);
    return jsonOutput({ result: 'ok' });
  }

  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME);
  sheet.appendRow([
    new Date(),
    data.page || '',
    data.email || '',
    JSON.stringify(data.categories || [], null, 0),
    data.notes || '',
    data.token || ''
  ]);

  if (data.email && data.code) {
    sendConfirmationEmail(data.email, data.code);
  }

  return jsonOutput({ result: 'ok' });
}

function sendConfirmationEmail(email, code) {
  var subject = '【GIFTED×GIFT】ご相談内容を受け付けました';
  var body =
    email + ' 様\n\n' +
    'この度はGIFTED×GIFTへご相談いただき、誠にありがとうございます。\n' +
    '内容を確認のうえ、担当より改めてご連絡させていただきます。\n\n' +
    '▼確認コード（次回のマイページログインにご利用いただけます）\n' +
    code + '\n\n' +
    '※このメールは送信専用です。ご返信いただいてもお答えできませんのでご了承ください。';

  MailApp.sendEmail(email, subject, body);
}

function doGet(e) {
  if (e.parameter.code && e.parameter.state) {
    return handleLineCallback(e.parameter.code, e.parameter.state);
  }

  if (e.parameter.action === 'myplans') {
    var email = (e.parameter.email || '').toLowerCase();
    if (!email) return jsonOutput({ error: 'email required' });

    var planSheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(PLAN_SHEET_NAME);
    var items = [];

    if (planSheet) {
      var planRows = planSheet.getDataRange().getValues();
      for (var p = 0; p < planRows.length; p++) {
        if (String(planRows[p][1]).toLowerCase() === email) {
          items.push({
            date: planRows[p][0],
            email: planRows[p][1],
            title: planRows[p][2],
            page: planRows[p][3],
            schedule: planRows[p][4],
            status: planRows[p][5]
          });
        }
      }
      items.reverse();
    }

    return jsonOutput(items);
  }

  if (!e.parameter.key || e.parameter.key !== SECRET) {
    return jsonOutput({ error: 'unauthorized' });
  }

  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME);
  var rows = sheet.getDataRange().getValues();
  var items = [];

  for (var i = 0; i < rows.length; i++) {
    items.push({
      date: rows[i][0],
      page: rows[i][1],
      email: rows[i][2],
      categories: rows[i][3],
      notes: rows[i][4]
    });
  }

  items.reverse();

  return jsonOutput(items);
}

function getOrCreateSheet(name) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName(name);
  if (!sheet) sheet = ss.insertSheet(name);
  return sheet;
}

function jsonOutput(obj) {
  return ContentService
    .createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}

// ▼LINE連携（スクリプトプロパティに以下4つのキーを設定してください。コード内には書きません）▼
// LINE_LOGIN_CHANNEL_ID / LINE_LOGIN_CHANNEL_SECRET / LINE_LOGIN_REDIRECT_URI / LINE_MSG_ACCESS_TOKEN

function handleLineCallback(code, token) {
  var props = PropertiesService.getScriptProperties();
  var loginChannelId = props.getProperty('LINE_LOGIN_CHANNEL_ID');
  var loginChannelSecret = props.getProperty('LINE_LOGIN_CHANNEL_SECRET');
  var redirectUri = props.getProperty('LINE_LOGIN_REDIRECT_URI');
  var msgAccessToken = props.getProperty('LINE_MSG_ACCESS_TOKEN');

  if (!loginChannelId || !loginChannelSecret || !redirectUri || !msgAccessToken) {
    return lineResultPage('LINE連携の設定が未完了です（スクリプトプロパティを確認してください）。');
  }

  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME);
  var rows = sheet.getDataRange().getValues();
  var rowIndex = -1;

  for (var i = 0; i < rows.length; i++) {
    if (rows[i][5] && String(rows[i][5]) === String(token)) {
      rowIndex = i;
      break;
    }
  }

  if (rowIndex === -1) {
    return lineResultPage('該当するご相談内容が見つかりませんでした。');
  }

  // 同じ相談内容につき、LINEへの送信は1回だけに制限する（戻る/進むによる再実行対策）
  if (rows[rowIndex][6]) {
    return lineResultPage('この内容はすでにLINEへ送信済みです。');
  }

  var tokenRes = UrlFetchApp.fetch('https://api.line.me/oauth2/v2.1/token', {
    method: 'post',
    payload: {
      grant_type: 'authorization_code',
      code: code,
      redirect_uri: redirectUri,
      client_id: loginChannelId,
      client_secret: loginChannelSecret
    },
    muteHttpExceptions: true
  });

  var tokenData = JSON.parse(tokenRes.getContentText());
  if (!tokenData.id_token) {
    return lineResultPage('LINEとの連携に失敗しました。もう一度お試しください。');
  }

  var userId = decodeLineUserId(tokenData.id_token);
  var summaryText = buildLineSummaryTextFromRow(rows[rowIndex]);

  UrlFetchApp.fetch('https://api.line.me/v2/bot/message/push', {
    method: 'post',
    contentType: 'application/json',
    headers: { Authorization: 'Bearer ' + msgAccessToken },
    payload: JSON.stringify({
      to: userId,
      messages: [{ type: 'text', text: summaryText }]
    }),
    muteHttpExceptions: true
  });

  sheet.getRange(rowIndex + 1, 7).setValue(new Date());

  return lineResultPage('ご相談内容をLINEに送信しました。LINEアプリのトークをご確認ください。');
}

function decodeLineUserId(idToken) {
  var payload = idToken.split('.')[1].replace(/-/g, '+').replace(/_/g, '/');
  while (payload.length % 4) payload += '=';
  var decoded = Utilities.newBlob(Utilities.base64Decode(payload)).getDataAsString();
  return JSON.parse(decoded).sub;
}

function buildLineSummaryTextFromRow(row) {
  var page = row[1];
  var notes = row[4];
  var categories = [];
  try { categories = JSON.parse(row[3]); } catch (err) { categories = []; }

  var lines = ['【GIFTED×GIFT】ご相談内容を受け付けました', '', 'ページ：' + page, ''];

  categories.forEach(function (cat) {
    lines.push('■ ' + cat.name);
    var fields = cat.fields || {};
    Object.keys(fields).forEach(function (key) {
      var value = fields[key];
      if (Array.isArray(value)) value = value.join('、');
      lines.push(key + '：' + value);
    });
    lines.push('');
  });

  if (notes) lines.push('補足：' + notes);

  return lines.join('\n');
}

function lineResultPage(message) {
  return HtmlService.createHtmlOutput(
    '<html><body style="font-family:sans-serif; text-align:center; padding:60px 20px; color:#333;">' +
    '<h2>' + message + '</h2>' +
    '</body></html>'
  );
}
